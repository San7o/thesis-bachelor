    every node/.style={

    },
    node distance=1.5cm,
    % Base Styles for Core Architecture Blocks
    block/.style={
      rectangle,
      draw=gray!80,
      fill=white,
      rounded corners=4px,
      minimum width=6cm,
      minimum height=1.2cm,
      font=\large\bfseries,
      align=center,
      line width=1.5pt
    },

      rectangle,
      draw=gray!50,

      align=center,
      line width=1pt
    },
    flow/.style={
      ->,
      >=stealth,
      line width=2pt,

    \node[block] (frontends) {Frontends};

    % Interfaces
    \node[container, below=of frontends, minimum width=13cm] (interfaces) {};
    \node[font=\large\bfseries, anchor=north, yshift=-5pt] at (interfaces.north)
    {User Interfaces};
    \node[inner_block] (syscall) at ($(interfaces.center) + (-4.5cm, -0.3cm)$) {System\\Calls};
    \node[inner_block] (pseudo)  at ($(interfaces.center) + (-1.5cm, -0.3cm)$) {Pseudo\\Filesystems};

    \node[font=\large\bfseries, anchor=north, yshift=-5pt] at (frameworks.north) {Tracing Frameworks};

    \node[inner_block] (perfevent)  at ($(frameworks.center) + (-0.0cm, 0.2cm)$) {perf};
    \node[inner_block] (ebpf)  at ($(frameworks.center) + (3.0cm,  0.2cm)$) {eBPF};
